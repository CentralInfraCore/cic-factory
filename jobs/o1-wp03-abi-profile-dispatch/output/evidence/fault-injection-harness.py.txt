def run(label, pkg, test):
    log = W / "evidence-raw" / "mut.run.log"
    subprocess.run([G, str(log), f"go test -count=1 -run '{test}' {pkg} 2>&1 | grep -E -- '--- FAIL|^(ok|FAIL)|\\[build failed\\]|expected|actual|Messages' | head -12"], check=False)
    return log.read_text()

lines.append("## baseline\n" + base)
body = "\n".join(l for l in base.splitlines() if not l.startswith("#"))
if "--- FAIL" in body or "FAIL\t" in body or "ok" not in body: print("baseline fails"); sys.exit(1)
for label, f, old, new, test in MUT:
    p = R / f; orig = p.read_text()
    if orig.count(old) != 1:
        lines.append(f"## {label}\n# RESULT: MUTATION NOT APPLIED (pattern count {orig.count(old)})\n"); bad += 1; continue
    try:
        p.write_text(orig.replace(old, new, 1))
        pkg = "./cmd/relay/" if f.startswith("cmd/") else "./core/cabinet/"
        out = run(label, pkg, test.split("/")[0])
        named = test.split("/")[-1] if "/" in test else test
        failed_named = f"--- FAIL: {test.split('/')[0]}" in out and ("[build failed]" not in out)
        sub_ok = (("/" not in test) or (named in out))
        status = "FAILED as required" if failed_named and sub_ok else "NOT CAUGHT / WRONG REASON"
        if status != "FAILED as required": bad += 1
        lines.append(f"## {label}\n# test: {test}\n# RESULT: {status}\n{out}")
    finally:
        p.write_text(orig)
