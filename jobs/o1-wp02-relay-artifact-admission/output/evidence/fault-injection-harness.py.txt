out = open(sys.argv[1], "w")
def log(s):
    out.write(s + "\n"); out.flush()

for m in M:
    name, path, old, new, tests, tags, reason = m[:7]
    extra = m[7] if len(m) > 7 else None
    p = R / path
    orig = p.read_text()
    assert orig.count(old) == 1, (name, "anchor not unique/absent")
    mutated = orig.replace(old, new)
    if extra and extra[0] != extra[1]:
        assert mutated.count(extra[0]) == 1, (name, "extra anchor")
        mutated = mutated.replace(extra[0], extra[1])
    p.write_text(mutated)
    try:
        tagflag = f"-tags {tags} " if tags else ""
        cmd = f"go test -count=1 {tagflag}-run '{tests}' ./cmd/relay/ 2>&1 | grep -E -- '--- FAIL|Error:|expected:|actual  :|Messages:|^(ok|FAIL)|_test.go:[0-9]+:|\\[build failed\\]|cannot|undefined' | head -60"
        r = subprocess.run([str(W / "gorun.sh"), str(W / "mut.tmp"), cmd], capture_output=True, text=True)
        body = (W / "mut.tmp").read_text()
        lines = [l for l in body.splitlines() if not l.startswith("# cmd:")]
        failed = any(l.lstrip().startswith("--- FAIL:") for l in lines)
        if any("[build failed]" in l for l in lines):
            failed = False
        log(f"## {name}\n# file: {path}\n# tests: {tests} {('(tags ' + tags + ')') if tags else ''}\n# expected named reason: {reason}\n# RESULT: {'FAILED as required' if failed else 'DID NOT FAIL or BUILD FAILED (not evidence)'}")
        log(body)
    finally:
        p.write_text(orig)
log("# all mutations restored")
