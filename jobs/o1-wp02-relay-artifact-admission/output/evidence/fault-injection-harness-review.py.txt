def run(pattern):
    log = W / "evidence-raw" / "mut2.run.log"
    subprocess.run([G, str(log), f"go test -count=1 -run '{pattern}' ./cmd/relay/ 2>&1 | grep -E -- '--- FAIL|^(ok|FAIL)|\\[build failed\\]' | head -12"], check=False)
    return log.read_text()
lines, bad = [], 0
base = run(PAT)
body = "\n".join(l for l in base.splitlines() if not l.startswith("#"))
lines.append("## baseline\n" + base)
if "--- FAIL" in body or "FAIL\t" in body or "ok" not in body: print("baseline fails"); sys.exit(1)
for label, f, old, new, test in MUT:
    p = R / f; orig = p.read_text()
    if orig.count(old) != 1:
        lines.append(f"## {label}\n# RESULT: MUTATION NOT APPLIED (pattern count {orig.count(old)})\n"); bad += 1; continue
    try:
        p.write_text(orig.replace(old, new, 1))
        out = run(PAT)
        top = test.split("/")[0]
        caught = f"--- FAIL: {top}" in out and "[build failed]" not in out
        status = "FAILED as required" if caught else "NOT CAUGHT / WRONG REASON"
        if not caught: bad += 1
        lines.append(f"## {label}\n# test: {test}\n# RESULT: {status}\n{out}")
    finally:
        p.write_text(orig)
after = run(PAT)
lines.append("## after restore\n" + after)
OUT.write_text("\n".join(lines))
print("mutations:", len(MUT), "not-as-required:", bad)
