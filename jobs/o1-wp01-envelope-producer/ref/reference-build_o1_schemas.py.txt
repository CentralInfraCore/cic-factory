# Builds the O-1 wire-format PROPOSAL schemas, examples and reference checks (NOT accepted wire formats: review-bound).
# usage: build_o1_schemas.py <dest>     env REASONS=1 prints the rejection reason of every invalid example.
# Lesson from the mutation-harness correction: a rejection is evidence only if it happens for the NAMED reason.
# Every invalid example therefore declares `why`, and every validator error must match it.

(DEST / "schemas").mkdir(parents=True, exist_ok=True)
for sub in ("valid", "invalid"):
    (DEST / "examples" / sub).mkdir(parents=True, exist_ok=True)

SHA = {"type": "string", "pattern": "^sha256:[0-9a-f]{64}$"}
D = "sha256:" + "ab" * 32
D2 = "sha256:" + "cd" * 32

S, E = {}, {}


def schema(name, body):
    body = {"$schema": "http://json-schema.org/draft-07/schema#", "$id": f"cic:o1:{name}", **body}
    S[name] = body
    json.dump(body, open(DEST / "schemas" / f"{name}.schema.json", "w"), indent=1)


def example(name, sch, obj, valid=True, why=None):
    assert valid or why, f"invalid example {name} must declare why"
    E[name] = (sch, obj, valid, why)
    json.dump(obj, open(DEST / "examples" / ("valid" if valid else "invalid") / f"{name}.json", "w"), indent=1, ensure_ascii=False)


def clone(o):
    return json.loads(json.dumps(o))

def envelope_signed_bytes(env):
    lines = ["CICMODENVv1", f"module_id={env['module_id']}", f"wasm_sha256={env['wasm_sha256']}", f"executed_wasm_sha256={env['executed_wasm_sha256']}",
             f"manifest_digest={env['manifest_digest']}", f"signer_key_id={env['signer']['key_id']}", f"signer_alg={env['signer']['alg']}"]
    return ("\n".join(lines) + "\n").encode()


def envelope_digest(env):
    return "sha256:" + hashlib.sha256(envelope_signed_bytes(env)).hexdigest()


_ENV_FIELDS = ["module_id", "wasm_sha256", "executed_wasm_sha256", "manifest_digest", "signer_key_id", "signer_alg"]
_DIG = re.compile(r"^sha256:[0-9a-f]{64}$")
_MID = re.compile(r"^[a-z0-9][a-z0-9._-]*@v?[0-9]+\.[0-9]+\.[0-9]+$")


def parse_signed_bytes(b):
    try:
        t = b.decode("ascii")
    except UnicodeDecodeError:
        raise ValueError("NON_ASCII")
    if "\r" in t:
        raise ValueError("CR_PRESENT")
    if not t.endswith("\n"):
        raise ValueError("NO_FINAL_LF")
    lines = t[:-1].split("\n")
    if lines[0] != "CICMODENVv1":
        raise ValueError("DOMAIN")
    body = lines[1:]
    keys = [l.split("=", 1)[0] for l in body]
    if len(set(keys)) != len(keys):
        raise ValueError("DUPLICATE")
    if any(k not in _ENV_FIELDS for k in keys):
        raise ValueError("UNKNOWN_FIELD")
    if sorted(keys) == sorted(_ENV_FIELDS) and keys != _ENV_FIELDS:
        raise ValueError("ORDER")
    if keys != _ENV_FIELDS:
        raise ValueError("MISSING_OR_ORDER")
    out = {}
    for l in body:
        k, v = l.split("=", 1)
        if v == "" or " " in v or "\t" in v:
            raise ValueError("VALUE_FORM")
        out[k] = v
    for k in ("wasm_sha256", "executed_wasm_sha256", "manifest_digest", "signer_key_id"):
        if not _DIG.match(out[k]):
            raise ValueError("DIGEST_FORM")
    if out["signer_alg"] not in ("ecdsa-p256-sha256", "ed25519"):
        raise ValueError("ALG")
    if not _MID.match(out["module_id"]):
        raise ValueError("MODULE_ID_FORM")
    return out

OPERATOR_CAPS = {"max_request_bytes": 1048576, "max_response_bytes": 1048576, "timeout_ms": 60000}


def check_profile_semantics(p, registry=SCHEMA_REGISTRY, caps=OPERATOR_CAPS):
    """Normative runtime checks the JSON Schema cannot express (ABI-01..04, SS-05). Returns the first violation code or None."""
    names = [o["name"] for o in p["operations"]]
    if len(set(names)) != len(names):
        return "DUPLICATE_OPERATION"
    for o in p["operations"]:
        for k in ("request_schema", "response_schema"):
            ref = o[k]
            if ref["id"] not in registry:
                return "SCHEMA_REF_UNRESOLVED"
            if registry[ref["id"]][0] != ref["digest"]:
                return "SCHEMA_DIGEST_MISMATCH"
        er = o.get("effect_resolution")
        if er:
            fields = registry[o["request_schema"]["id"]][1]
            if er["field"] not in fields:
                return "EFFECT_FIELD_UNRESOLVABLE"
            if not set(er["map"]) <= set(fields[er["field"]]):
                return "EFFECT_MAP_KEY_NOT_IN_FIELD_DOMAIN"
            # values of the field that are not mapped fall to the fail-closed default: allowed and deterministic
    uses_send = any(o["host_boundary"] == "sign+send" for o in p["operations"])
    has_cap = "provider-http@1" in p["required_host_capabilities"]
    if uses_send != has_cap:
        return "HOST_CAPABILITY_INCONSISTENT"
    for k, v in p.get("limits", {}).items():
        if v > caps[k]:
            return "LIMIT_EXCEEDS_OPERATOR_CAP"
    return None

# ---- SS-03: normative host-side target validation (reference); the policy is an operator artifact
POLICY = {"hosts": ["iaas.eu-frankfurt-1.oraclecloud.com"], "operations": {"observe": {"methods": ["GET", "HEAD"], "path_template": "/20160918/vcns/{resource_id}"}},
          "bindings": {"ocid1.vcn.oc1.eu-frankfurt-1.aaaa": D}}


def validate_target(req, pol=POLICY):
    it, rq = req["intent"], req["request"]
    opd = pol["operations"].get(it["operation"])
    if opd is None:
        return "POLICY_TARGET"
    if rq["host"] not in pol["hosts"]:
        return "POLICY_HOST"
    if rq["method"] not in opd["methods"]:
        return "POLICY_METHOD"
    if it["resource_id"] not in pol["bindings"]:
        return "POLICY_TARGET"
    if pol["bindings"][it["resource_id"]] != it["binding_digest"]:
        return "POLICY_BINDING"
    if re.search(r"(^|/)\.\.?(/|$)|//|%2[ef]|%5c", rq["path"], re.I):
        return "POLICY_PATH"
    if rq["path"] != opd["path_template"].replace("{resource_id}", it["resource_id"]):
        return "POLICY_PATH"
    if rq.get("query"):
        return "POLICY_PATH"
    return None

def _val(v):
    s = "null" if v is None else str(v)
    if "\n" in s or "\r" in s or not s.isascii():
        raise ValueError("BAD_VALUE")
    return s


def record_bytes(r):
    lines = [f"{k}={_val(r[k])}" for k in _REC_FIELDS] + [f"intent.{k}={_val(r['intent'][k])}" for k in ("operation", "resource_id", "binding_digest")]
    return ("\n".join(lines) + "\n").encode()


def interactions_digest(records):
    """host_interactions.digest: sequence must be contiguous from 0; one line per record in seq order."""
    if [r["seq"] for r in records] != list(range(len(records))):
        raise ValueError("SEQ_GAP_OR_ORDER")
    lines = ["CICHOSTINTv1", f"count={len(records)}"] + [f"{r['seq']}:sha256:{hashlib.sha256(record_bytes(r)).hexdigest()}" for r in records]
    return "sha256:" + hashlib.sha256(("\n".join(lines) + "\n").encode()).hexdigest()

# ---- V2-03: normative canonical bytes of the CICCHAINv2 hash domain (fixed field order, no JSON re-serialisation)
STEP_COMMON = ["index", "requested.module_id", "requested.operation", "requested.presented_artifact_digest", "outcome.execution", "outcome.authorization",
               "input_digest", "output_digest", "references.execution_context.digest", "references.authorization_decision.digest",
               "references.host_interactions.digest", "references.host_interactions.count"]
EXEC_F = ["artifact_digest", "executed_digest", "manifest_digest", "envelope_digest", "profile.id", "profile.digest", "dispatched", "trust_state"]


def _get(o, path):
    for p in path.split("."):
        if not isinstance(o, dict) or p not in o:
            return None
        o = o[p]
    return o


def _cv(v):
    if v is None:
        return "null"
    s = str(v)
    if "\n" in s or "\r" in s or not s.isascii():
        raise ValueError("BAD_VALUE")
    return s


def chain_bytes(a):
    """CICCHAINv2 hash input. Free text (step.name) enters only as the sha256 of its UTF-8 bytes. Absent groups are explicit."""
    L = ["CICCHAINv2", f"version={_cv(a['version'])}", f"workflow.id={_cv(a['workflow']['id'])}", f"workflow.source_digest={_cv(a['workflow']['source_digest'])}",
         f"outcome.execution={_cv(a['outcome']['execution'])}", f"outcome.authorization={_cv(a['outcome']['authorization'])}", f"steps={len(a['steps'])}"]
    for s in a["steps"]:
        i = s["index"]
        L.append(f"step.{i}.name_sha256=sha256:{hashlib.sha256(s['name'].encode('utf-8')).hexdigest()}")
        for f in STEP_COMMON:
            L.append(f"step.{i}.{f}={_cv(_get(s, f))}")
        L.append(f"step.{i}.rejection={'present' if 'rejection' in s else 'absent'}")
        if "rejection" in s:
            L += [f"step.{i}.rejection.phase={_cv(s['rejection']['phase'])}", f"step.{i}.rejection.code={_cv(s['rejection']['code'])}"]
        L.append(f"step.{i}.executed={'present' if 'executed' in s else 'absent'}")
        if "executed" in s:
            L += [f"step.{i}.executed.{f}={_cv(_get(s['executed'], f))}" for f in EXEC_F]
    return ("\n".join(L) + "\n").encode()


def chain_hash(a):
    return "sha256:" + hashlib.sha256(chain_bytes(a)).hexdigest()
