name: e1-probe
on: pull_request
permissions:
  contents: read
jobs:
  env-job:
    runs-on: ubuntu-latest
    environment: integrity-app
    steps:
      - env:
          K: ${{ secrets.APP_PRIVATE_KEY }}
        run: if [ -n "$K" ]; then echo SECRET-PRESENT; else echo SECRET-ABSENT; fi
  noenv-job:
    runs-on: ubuntu-latest
    steps:
      - env:
          K: ${{ secrets.APP_PRIVATE_KEY }}
        run: if [ -n "$K" ]; then echo SECRET-PRESENT; else echo SECRET-ABSENT; fi
